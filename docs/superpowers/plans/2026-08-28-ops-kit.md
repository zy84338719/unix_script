> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增 `sys-tools/ops-kit` 运维工具箱——inspect 聚合巡检 + log/svc/audit 三组运维子命令（spec: `docs/superpowers/specs/2026-08-28-ops-kit-design.md`）。

**Architecture:** 单文件 bash 模块（学 sys-cmd 多子命令 + disk 的护栏/stub 测试套路）。解析与分级全部收进 `_ops_*` 纯函数（可在 macOS 上单测），命令层只做数据采集（df/ss/systemctl/journalctl 等）+ 渲染。source 守护使单测能复用纯函数；写操作依赖 lib/common.sh 的 dry-run sudo 遮蔽 + 交互确认。

**Tech Stack:** Bash 3.2+ / lib/common.sh（header/info/error/emit_status/sudo 遮蔽）/ systemd 工具链（运行时可选）。

## Global Constraints

- 平台：Linux-only。macOS 执行任何子命令输出"本模块仅支持 Linux"退出 0；**source 时**不做平台退出（守护只拦执行分发）。
- 分级阈值（verbatim）：磁盘 80⚠️/90🔴；journal 2G⚠️；更新 >0⚠️ >50🔴；MaxAuthTries ≤5 ok / 6-10 ⚠️ / >10 🔴。
- 只读命令退出码恒 0；单项检查失败记"跳过"不中止。
- 写操作（vacuum/rotate apply/svc stop|restart|disable）三件套：显式参数必填 + 交互终端确认（`[ -t 0 ]`）+ 全局 `--dry-run`（source common.sh 后 sudo 已被遮蔽，直接调 `sudo`）。
- 机器契约：`status` 用 `emit_status`；`inspect --json` 输出合法 JSON（单行）。
- `set -euo pipefail`；命名 `_ops_*`（纯函数）/ `cmd_*`（命令）；采集命令一律 `command -v` 预检。
- 纯函数不得调用采集命令（参数进、stdout 出），保证 macOS 可单测。

```bash
#!/usr/bin/env bash
#
# sys-tools/ops-kit/install.sh
#
# 运维工具箱 —— inspect 巡检聚合 + log/svc/audit 三组运维（仅 Linux）。
# 只读优先；写操作 = 显式参数 + 交互确认 + 全局 --dry-run。
#
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
# shellcheck source=../../lib/common.sh
source "$SCRIPT_DIR/../../lib/common.sh"

OPS_OK='✅'; OPS_WARN='⚠️ '; OPS_CRIT='🔴'; OPS_SKIP='⬜'

# ---- 平台（仅执行路径；source 时不触发）----
_ops_platform_check() {
    if [[ "${OS_TYPE:-}" == "darwin" ]]; then
        info "ops-kit 仅支持 Linux（依赖 systemd），macOS 不适用。"
        exit 0
    fi
}

# ---- 纯函数区（Task 2 填充）----
# _ops_size_to_mb, _ops_df_parse, _ops_journal_usage_parse, _ops_ss_parse,
# _ops_lastb_count, _ops_failed_parse, _ops_sshd_check, _ops_updates_count, _ops_json_escape

# ---- 状态 ----
_ops_tool_avail() { command -v "$1" >/dev/null 2>&1 && echo 1 || echo 0; }

_ops_state() {
    if [[ "${OS_TYPE:-}" == "darwin" ]]; then echo "n/a"; return; fi
    echo "installed"   # 纯命令封装，无安装物
}

status_ops_kit() {
    local state; state=$(_ops_state)
    if [[ "$state" == "n/a" ]]; then
        emit_status "n/a" "⏭️  ops-kit 运维工具箱（仅 Linux，本机不适用）"
        return 0
    fi
    emit_status "installed" "✅ ops-kit 运维工具箱就绪（inspect 巡检 / log 日志 / svc 服务 / audit 基线）"
    emit_extra "tools=systemctl:$(_ops_tool_avail systemctl),journalctl:$(_ops_tool_avail journalctl),smartctl:$(_ops_tool_avail smartctl),lastb:$(_ops_tool_avail lastb),ss:$(_ops_tool_avail ss)"
}

cmd_install() {
    _ops_platform_check
    local t missing=0
    for t in systemctl journalctl ss logrotate; do
        if command -v "$t" >/dev/null 2>&1; then info "✔ $t 可用"; else warn "✘ $t 缺失（相关子命令将降级）"; missing=$((missing+1)); fi
    done
    if (( missing == 0 )); then info "🎉 依赖齐备，试试: ./install.sh ops-kit inspect"; else info "缺失 $missing 项；缺啥用啥，不影响其余子命令"; fi
    return 0
}

cmd_uninstall() { info "ops-kit 为只读命令封装，不安装任何组件，无需卸载"; }

usage() {
    cat <<'EOF'
运维工具箱（仅 Linux）—— 一站式巡检与日常运维

用法: ops-kit {inspect|log|svc|audit|status|help} [参数...]

  inspect [--json]        聚合巡检：磁盘/SMART/失败服务/journal/SSH 基线/公网端口/更新/登录失败
  log status              journal 占用 + /var/log 大户 TOP5
  log vacuum <200M|2weeks>  清理 journald（写操作，需交互确认）
  log rotate list|show <app>|apply <app>   查看/生成 /etc/logrotate.d 条目
  svc failed              systemd 失败单元汇总（unit/since/journal 尾行）
  svc logs <unit> [n]     查看单元日志（默认 50 行）
  svc status|start|stop|restart|enable|disable <unit>   systemctl 透传
  audit ssh|ports|updates|all   安全基线自查（只读）
  status                  依赖齐备度（UXS_STATUS_MODE=machine 可读）
  help                    本帮助
EOF
}

# ---- 分发 ----
main() {
    local sub="${1:-inspect}"
    shift || true
    case "$sub" in
        inspect) _ops_platform_check; cmd_inspect "$@" ;;
        log)     _ops_platform_check; cmd_log "$@" ;;
        svc)     _ops_platform_check; cmd_svc "$@" ;;
        audit)   _ops_platform_check; cmd_audit "$@" ;;
        install) cmd_install ;;
        uninstall) cmd_uninstall ;;
        status)  status_ops_kit ;;
        help|-h|--help) usage ;;
        *) error "未知子命令: $sub（可用: inspect/log/svc/audit/status/help）"; usage; return 1 ;;
    esac
}

# source 守护：被 source（单测）时只提供函数，不执行分发
if [[ "${BASH_SOURCE[0]}" == "$0" ]]; then
    main "$@"
fi
```

```bash
_ops_size_to_mb() {
    local s="${1:-}" num unit
    [[ "$s" =~ ^([0-9]+(\.[0-9]+)?)([KMGTP]?)(B?)$ ]] || { echo 0; return; }
    num="${BASH_REMATCH[1]}"; unit="${BASH_REMATCH[3]}"
    case "$unit" in
        ""|B) num=$(awk "BEGIN{printf \"%d\", $num/1048576}") ;;
        K)    num=$(awk "BEGIN{printf \"%d\", $num/1024}") ;;
        M)    num=$(awk "BEGIN{printf \"%d\", $num}") ;;
        G)    num=$(awk "BEGIN{printf \"%d\", $num*1024}") ;;
        T)    num=$(awk "BEGIN{printf \"%d\", $num*1048576}") ;;
        P)    num=$(awk "BEGIN{printf \"%d\", $num*1073741824}") ;;
    esac
    echo "$num"
}

_ops_df_parse() {
    echo "$1" | awk 'NR>1 && $1 !~ /^(tmpfs|devtmpfs|overlay|loop|squashfs|udev)$/ {gsub(/%/,"",$5); print $6"|"$5}' | grep -v '|$' || true
}

_ops_journal_usage_parse() {
    # "Archived and active journals take up 3.9G in the file system." → 3.9G
    echo "$1" | grep -oE 'take up [0-9]+(\.[0-9]+)?[KMGT]' | awk '{print $3}' || true
}

_ops_ss_parse() {
    echo "$1" | awk 'NR>1 && $2=="LISTEN" {
        proc=$NF; sub(/^users:\(\("/,"",proc); sub(/",.*/,"",proc);
        print $1"|"$4"|"(proc==""?"unknown":proc)}' || true
}

_ops_is_public_addr() {
    case "$1" in
        127.*|::1*|\[::1\]*) return 1 ;;
        0.0.0.0*|\[::\]*|\*:*|10.*|172.1[6-9].*|172.2[0-9].*|172.3[0-1].*|192.168.*|169.254*|fe80*|\[fe80*) return 1 ;;
        *) return 0 ;;
    esac
}
# 注：私网段同样返回 1（非公网暴露）；无格式匹配的兜底视为公网（保守告警）

_ops_lastb_count() {
    printf '%s\n' "$1" | grep -cvE '^\s*$|^btmp begins' || true
}

_ops_failed_units() {
    printf '%s\n' "$1" | awk 'NF && $1 ~ /\.service$|\.timer$|\.socket$|\.mount$/ {print $1}' || true
}

_ops_sshd_check() {
    local text="$1" key val
    # sshd -T 输出小写无冒号；sshd_config 可能有注释/大小写。统一小写后逐键找。
    local low; low=$(printf '%s\n' "$text" | tr '[:upper:]' '[:lower:]')
    _ops_sshd_one() {  # <text> <键>  → "值" 或空
        printf '%s\n' "$1" | awk -v k="$2" '$1==k{print $2; exit}'
    }
    val=$(_ops_sshd_one "$low" permitrootlogin)
    case "${val:-unset}" in
        prohibit-password|no|unset) echo "ok|permitrootlogin=${val:-unset}" ;;
        yes) echo "crit|permitrootlogin=yes" ;;
        *)   echo "warn|permitrootlogin=$val" ;;
    esac
    val=$(_ops_sshd_one "$low" passwordauthentication)
    case "${val:-no}" in
        no|unset) echo "ok|passwordauthentication=${val:-unset}" ;;
        yes) echo "warn|passwordauthentication=yes" ;;
        *)   echo "warn|passwordauthentication=$val" ;;
    esac
    val=$(_ops_sshd_one "$low" maxauthtries)
    if [[ -z "$val" ]]; then echo "ok|maxauthtries=unset"
    elif (( val <= 5 )); then echo "ok|maxauthtries=$val"
    elif (( val <= 10 )); then echo "warn|maxauthtries=$val"
    else echo "crit|maxauthtries=$val"; fi
    val=$(_ops_sshd_one "$low" x11forwarding)
    case "${val:-no}" in
        no|unset) echo "ok|x11forwarding=${val:-unset}" ;;
        *) echo "warn|x11forwarding=$val" ;;
    esac
}

_ops_updates_count() {
    local n; n=$(printf '%s\n' "$1" | grep -oE '[0-9]+' | head -1)
    echo "${n:-0}"
}

cmd_inspect() {
    local json=0; [[ "${1:-}" == "--json" ]] && json=1
    OPS_CHECKS=''; OPS_JSON=''; OPS_CRIT_N=0; OPS_WARN_N=0

    # 1) 磁盘空间

    while IFS= read -r dline; do
        [[ -z "$dline" ]] && continue
        mount="${dline%%|*}"; pct="${dline##*|}"
        if (( pct > worst )); then worst=$pct; wline="$mount"; fi
        if (( pct >= 90 )); then _ops_check_add crit "磁盘空间 $mount" "${pct}% 已用"
        elif (( pct >= 80 )); then _ops_check_add warn "磁盘空间 $mount" "${pct}% 已用"; fi
    done < <(_ops_df_parse "$(df -P 2>/dev/null)")
    (( worst > 0 && worst < 80 )) && _ops_check_add ok "磁盘空间" "最大使用率 ${worst}%（$wline）"

    if command -v smartctl >/dev/null 2>&1; then
        local dev h
        for dev in /dev/sd? /dev/nvme?; do
            [[ -e "$dev" ]] || continue
            h=$(sudo -n smartctl -H "$dev" 2>/dev/null || smartctl -H "$dev" 2>/dev/null || true)
            if printf '%s' "$h" | grep -q PASSED; then _ops_check_add ok "SMART $dev" "PASSED"
            elif [[ -z "$h" ]]; then _ops_check_add skip "SMART $dev" "读取失败（权限/USB 桥）"
            else _ops_check_add crit "SMART $dev" "总评非 PASSED，运行: sudo smartctl -H $dev"; fi
        done
    else
        _ops_check_add skip "SMART" "未安装 smartmontools"
    fi

    if command -v systemctl >/dev/null 2>&1 && systemctl is-system-running >/dev/null 2>&1 || [[ "$(systemctl is-system-running 2>/dev/null || true)" == "degraded" ]]; then

    else

    fi

        else _ops_check_add ok "journal 占用" "${jmb}MB"; fi
    else
        _ops_check_add skip "journal" "journalctl 不可用"
    fi

    # 5) SSH 基线摘要
    local sshout='' sshlvl=ok
    if command -v sshd >/dev/null 2>&1; then

    fi
    if [[ -n "$sshout" ]]; then
        local sline
        while IFS= read -r sline; do
            case "${sline%%|*}" in crit) sshlvl=crit;; warn) [[ "$sshlvl" != crit ]] && sshlvl=warn;; esac
        done < <(_ops_sshd_check "$sshout")

    else
        _ops_check_add skip "SSH 基线" "读不到 sshd 配置"
    fi

    if command -v ss >/dev/null 2>&1; then
        local sp pub='' sline2
        while IFS= read -r sp; do
            [[ -z "$sp" ]] && continue
            sline2="${sp#*|}"; sline2="${sline2%%|*}"
            if _ops_is_public_addr "$sline2"; then pub+="${sp}"$'\n'; fi
        done < <(_ops_ss_parse "$(ss -tulpn 2>/dev/null || true)")
        if [[ -z "$pub" ]]; then _ops_check_add ok "公网监听" "无 0.0.0.0/[::] 监听"
        else _ops_check_add warn "公网监听" "$(printf '%s' "$pub" | wc -l | tr -d ' ') 个，详见 ./install.sh ops-kit audit ports"; fi
    else
        _ops_check_add skip "公网监听" "ss 不可用"
    fi

    # 7) 待更新
    local updc=0
    if command -v apt-get >/dev/null 2>&1; then updc=$(_ops_updates_count "$(apt-get -s upgrade 2>/dev/null | grep -c '^Inst' || true)")
    elif command -v dnf >/dev/null 2>&1; then updc=$(_ops_updates_count "$(dnf -q check-update 2>/dev/null | grep -cE '^\S+\s' || true)")
    elif command -v yum >/dev/null 2>&1; then updc=$(_ops_updates_count "$(yum -q check-update 2>/dev/null | grep -cE '^\S+\s' || true)")
    elif command -v pacman >/dev/null 2>&1; then updc=$(_ops_updates_count "$(pacman -Qu 2>/dev/null | grep -c . || true)")
    elif command -v apk >/dev/null 2>&1; then updc=$(_ops_updates_count "$(apk version -l '<' 2>/dev/null | grep -c '^<' || true)") ; fi
    if   (( updc > 50 )); then _ops_check_add crit "待更新" "${updc} 个"
    elif (( updc > 0 )); then _ops_check_add warn "待更新" "${updc} 个"
    else _ops_check_add ok "待更新" "0"; fi

    # 8) 登录失败
    if command -v lastb >/dev/null 2>&1 && [[ -e /var/log/btmp ]]; then
        local lbc; lbc=$(_ops_lastb_count "$(lastb -n 50 2>/dev/null | head -50 || true)")
        if (( lbc > 0 )); then _ops_check_add warn "登录失败" "近期 ${lbc}+ 条（lastb）"
        else _ops_check_add ok "登录失败" "无记录"; fi
    else
        _ops_check_add skip "登录失败" "lastb/btmp 不可用"
    fi

    # 输出
    if (( json )); then
        printf '{"summary":{"crit":%d,"warn":%d},"checks":[%s]}\n' "$OPS_CRIT_N" "$OPS_WARN_N" "$OPS_JSON"
    else
        header "🩺 系统巡检报告（$(date '+%F %T')）"
        printf '%s' "$OPS_CHECKS"

        elif (( OPS_WARN_N > 0 )); then echo "${OPS_WARN}巡检汇总：${OPS_WARN_N} 项注意，无危险项"
        else echo "${OPS_OK}巡检汇总：全部通过"; fi
    fi
    return 0
}

        elif command -v yum >/dev/null 2>&1; then src=yum; n=$(_ops_updates_count "$(yum -q check-update 2>/dev/null | grep -cE '^\S+\s' || true)")
        elif command -v pacman >/dev/null 2>&1; then src=pacman; n=$(_ops_updates_count "$(pacman -Qu 2>/dev/null | grep -c . || true)")
        elif command -v apk >/dev/null 2>&1; then src=apk; n=$(_ops_updates_count "$(apk version -l '<' 2>/dev/null | grep -c '^<' || true)")
        else info "未识别包管理器，跳过"; return 0; fi
        info "[$src] 待更新 ${n} 个；自动安装指路: ./install.sh sys-setup unattended"
    }
    case "$only" in
        ssh) _do_ssh ;;
        ports) _do_ports ;;
        updates) _do_updates ;;
        all) _do_ssh; _do_ports; _do_updates ;;
        *) error "用法: ops-kit audit ssh|ports|updates|all"; return 1 ;;
    esac
    return 0
}
```
