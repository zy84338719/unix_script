# CLI v2 分组子命令面 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** unix_script 的 CLI 一次性切换到「分组子命令面」，旧扁平 flag 全部删除（无兼容期），并同步文档/补全/CI/快照基线。

**Architecture:** 入口 `install.sh` 的分发层从「扁平 flag 的 `case "$1"`」改成两层：先认**命令名**（保留词表），首参非保留词则按**模块直通**；每个命令有自己的 flag 集合，解析后转发给既有 `lib/` 呈现函数。呈现函数的**输出格式逐字不动**（快照门禁能证明），只换入口名与参数形式。

**Tech Stack:** bash 3.2+（macOS 自带 /bin/bash 为基线）、shellkit（`vendor/shellkit/` 副本）、shellcheck、自研 `tests/ci_run.sh` 三阶段门禁 + `tools/check_snapshot.sh` 行为快照门禁。

## Global Constraints

- **bash 3.2 基线**：禁 `declare -A` / `${v^^}` / `${v,,}` / `mapfile` / `[[ -v ]]` / nameref；空数组展开一律 `${arr[@]+"${arr[@]}"}`。
- **stdout 只给数据，日志一律 stderr**；出参用 `printf -v`。
- **退出码口径**（AGENTS.md 已写明）：`0` 成功（状态类子命令恒 0）｜ `1` 一般错误 ｜ `2` 用法错（参数不对/未知选项/未知子命令）｜ `3` 前置不满足（模块不存在/平台不适用/缺命令）｜ `4` 网络失败 ｜ `5` 校验失败。
- **平台可见性**：默认只显示本机适用模块；`--all` = 含平台不适用模块（等价旧 `UNIX_SCRIPT_SHOW_ALL=1`，该环境变量继续有效）。
- **快照门禁**：改了 CLI 输出必须 `bash tools/snapshot.sh` 重生成基线 **且** `bash tools/snapshot.sh --approve` 更新登记，两处进同一 diff。
- **缩进**：unix_script 仓用 4 空格；CI 不跑 shfmt（该仓无 fmt 门禁），跟随既有行文风格即可。
- **禁止 `git add -A` / `git add .`**：必须显式列文件。
- **基线是 macOS 的 `/bin/bash`（3.2）**：验 3.2 必须显式 `/bin/bash`，不能靠登录 shell 里的 brew bash 5.x。

```bash
#!/usr/bin/env bash
#
# lib/cli_v2.sh
#
# CLI v2：分组子命令面。
#
# 分发规则（两层）：
#   1) 首参是保留词 → 当命令解析，命令各有自己的 flag 集合；
#   2) 首参不是保留词 → 当模块名直通（uxs <mod> [subcmd...]）。
#
# 旧扁平 flag 一次性删除：命中即退出 2 并给一行迁移指引（不是兼容期，命令照样失败）。
#
# 退出码口径：0 成功 / 1 一般错误 / 2 用法错 / 3 前置不满足 / 4 网络失败 / 5 校验失败。

if [[ -n "${_CLI_V2_LOADED:-}" ]]; then return 0 2>/dev/null || exit 0; fi
_CLI_V2_LOADED=1

# 保留词表：首参命中即当命令，不当模块名（78 个模块已核对无撞名）。
CLI_V2_COMMANDS="install remove status list search info doctor export apply update scaffold completions cli help version"

# 旧 flag → 新写法（迁移指引用）
CLI_V2_MIGRATIONS="--list:uxs list
--list-modules:uxs list --full
--list-categories:uxs list --grouped
--status:uxs status
--status-json:uxs status --json
check-update:uxs update --check
export-profile:uxs export
apply-profile:uxs apply
install-cli:uxs cli install
remove-cli:uxs cli uninstall"

# cli_v2_is_command <word> — 是否保留词
cli_v2_is_command() {
    local w="${1:-}" c
    for c in $CLI_V2_COMMANDS; do
        [[ "$w" == "$c" ]] && return 0
    done
    return 1
}

# cli_v2_migration_for <old> — 查迁移指引；查不到返回 1
cli_v2_migration_for() {
    local old="${1:-}" line k v
    for line in $CLI_V2_MIGRATIONS; do
        k="${line%%:*}"
        v="${line#*:}"
        [[ "$k" == "$old" ]] && { printf '%s' "$v"; return 0; }
    done
    return 1
}

# cli_v2_die_old <old> — 旧写法命中：报错 + 迁移指引 + 退出 2
cli_v2_die_old() {
    local old="${1:-}" hint
    hint=$(cli_v2_migration_for "$old" || true)
    error "未知选项: $old"
    if [[ -n "$hint" ]]; then
        info "该写法已删除，请改用: $hint"
    fi
    exit "$SK_USAGE"
}
```

```bash
# cli_v2_dispatch <args...> — CLI v2 分发；内部 exit，不 return
cli_v2_dispatch() {
    if [[ $# -eq 0 ]]; then
        show_usage
        exit "$SK_USAGE"
    fi
    local first="$1"
    # 顶层 flag：只保留 --help / --version / --list-all
    case "$first" in
        -h|--help)   show_usage; exit 0 ;;
        -v|--version) echo "unix_script $(cat "$SCRIPT_DIR/VERSION" 2>/dev/null || echo unknown)"; exit 0 ;;
        --list-all)  cli_v2_list --full --all; exit $? ;;
        -*)          cli_v2_die_old "$first" ;;
    esac
    if cli_v2_is_command "$first"; then
        shift
        case "$first" in
            install)  cli_v2_install "$@" ;;
            remove)   cli_v2_remove "$@" ;;
            status)   cli_v2_status "$@" ;;
            list)     cli_v2_list "$@" ;;
            search)   cli_v2_search "$@" ;;
            info)     cli_v2_info "$@" ;;
            doctor)   run_doctor; exit $? ;;
            export)   cli_v2_export "$@" ;;
            apply)    cli_v2_apply "$@" ;;
            update)   cli_v2_update "$@" ;;
            scaffold) cli_v2_scaffold "$@" ;;
            completions) cli_v2_completions "$@" ;;
            cli)      cli_v2_cli "$@" ;;
            help)     show_usage; exit 0 ;;
            version)  echo "unix_script $(cat "$SCRIPT_DIR/VERSION" 2>/dev/null || echo unknown)"; exit 0 ;;
        esac
        exit "$SK_USAGE"
    fi
    # 非保留词 → 模块直通
    dispatch_module_or_passthrough "$@"
    exit $?
}
```

```bash
# 解析通用 flag：--all / --json / --dry-run / --no-deps / --force / --out / --category
# 用法：cli_v2_take_flag <flag> <"$@" 的剩余> —— bash 3.2 不能改 "$@"，故用全局数组
CLI_V2_ARGS=()

cli_v2_collect() {
    CLI_V2_ARGS=()
    local a
    for a in "$@"; do
        CLI_V2_ARGS[${#CLI_V2_ARGS[@]}]="$a"
    done
}

# cli_v2_list [--full] [--grouped] [--category <cat>] [--all]
cli_v2_list() {
    local full=0 grouped=0 category="" show_all=0
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --full)    full=1; shift ;;
            --grouped) grouped=1; shift ;;
            --category) category="${2:-}"; shift 2 ;;
            --all)     show_all=1; shift ;;
            *) error "未知选项: $1"; exit "$SK_USAGE" ;;
        esac
    done
    [[ "$show_all" -eq 1 ]] && export UNIX_SCRIPT_SHOW_ALL=1
    if [[ "$full" -eq 1 ]]; then
        show_list_modules
    elif [[ "$grouped" -eq 1 ]]; then
        show_list_categories "$category"
    elif [[ -n "$category" ]]; then
        show_list_categories "$category"
    else
        local mod
        for mod in $(registry_visible_modules); do printf '%s ' "$mod"; done
        echo
    fi
}

# cli_v2_status [mod...] [--json] [--all]
cli_v2_status() {
    local json=0 show_all=0
    local -a mods=()
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --json) json=1; shift ;;
            --all)  show_all=1; shift ;;
            -*) error "未知选项: $1"; exit "$SK_USAGE" ;;
            *) mods[${#mods[@]}]="$1"; shift ;;
        esac
    done
    [[ "$show_all" -eq 1 ]] && export UNIX_SCRIPT_SHOW_ALL=1
    if [[ "$json" -eq 1 ]]; then
        show_status_json
        return 0
    fi
    if [[ "${#mods[@]}" -eq 0 ]]; then
        show_installed_services
        return 0
    fi
    local mod rc=0
    for mod in "${mods[@]}"; do
        dispatch_module_or_passthrough "$mod" status || rc=$?
    done
    return "$rc"
}
```

```bash
# cli_v2_install <mod>... [--no-deps] [--dry-run]
cli_v2_install() {
    local dry=0 nodeps=0
    local -a mods=()
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --dry-run) dry=1; shift ;;
            --no-deps) nodeps=1; shift ;;
            -*) error "未知选项: $1"; exit "$SK_USAGE" ;;
            *) mods[${#mods[@]}]="$1"; shift ;;
        esac
    done
    if [[ "${#mods[@]}" -eq 0 ]]; then
        show_usage
        exit "$SK_USAGE"
    fi
    [[ "$dry" -eq 1 ]] && { export UNIX_SCRIPT_DRY_RUN=1; uxs_install_sudo_shim; }
    [[ "$nodeps" -eq 1 ]] && export UNIX_SCRIPT_NO_DEPS=1
    local mod rc=0
    for mod in "${mods[@]}"; do
        dispatch_module "$mod" || rc=$?
    done
    return "$rc"
}

# cli_v2_remove <mod>...
cli_v2_remove() {
    local -a mods=()
    while [[ $# -gt 0 ]]; do
        case "$1" in
            -*) error "未知选项: $1"; exit "$SK_USAGE" ;;
            *) mods[${#mods[@]}]="$1"; shift ;;
        esac
    done
    if [[ "${#mods[@]}" -eq 0 ]]; then
        show_usage
        exit "$SK_USAGE"
    fi
    local mod rc=0
    for mod in "${mods[@]}"; do
        dispatch_module_or_passthrough "$mod" uninstall || rc=$?
    done
    return "$rc"
}

# cli_v2_export [--out FILE]
cli_v2_export() {
    local out=""
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --out) out="${2:-}"; shift 2 ;;
            -*) error "未知选项: $1"; exit "$SK_USAGE" ;;
            *) error "未知选项: $1"; exit "$SK_USAGE" ;;
        esac
    done
    if [[ -n "$out" ]]; then
        export_profile "$out"
    else
        export_profile
    fi
    return $?
}

# cli_v2_apply <profile> [--force] [--dry-run]
cli_v2_apply() {
    local force=0 dry=0 file=""
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --force)  force=1; shift ;;
            --dry-run) dry=1; shift ;;
            -*) error "未知选项: $1"; exit "$SK_USAGE" ;;
            *) file="$1"; shift ;;
        esac
    done
    [[ -n "$file" ]] || { error "用法: uxs apply <profile> [--force] [--dry-run]"; exit "$SK_USAGE"; }
    local -a extra=()
    [[ "$force" -eq 1 ]] && extra[${#extra[@]}]="--force"
    [[ "$dry" -eq 1 ]] && extra[${#extra[@]}]="--dry-run"
    apply_profile "$file" ${extra[@]+"${extra[@]}"}
    return $?
}

# cli_v2_update [--check]
cli_v2_update() {
    local check=0
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --check) check=1; shift ;;
            -*) error "未知选项: $1"; exit "$SK_USAGE" ;;
            *) error "未知选项: $1"; exit "$SK_USAGE" ;;
        esac
    done
    if [[ "$check" -eq 1 ]]; then
        check_for_update
        return $?
    fi
    do_self_update
    return $?
}

# cli_v2_scaffold <mod> [--category <cat>] [--label <label>]
cli_v2_scaffold() {
    local mod="" category="" label=""
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --category) category="${2:-}"; shift 2 ;;
            --label)    label="${2:-}"; shift 2 ;;
            -*) error "未知选项: $1"; exit "$SK_USAGE" ;;
            *) mod="$1"; shift ;;
        esac
    done
    [[ -n "$mod" ]] || { error "用法: uxs scaffold <mod> [--category <分类>] [--label <标签>]"; exit "$SK_USAGE"; }
    if [[ -n "$category" ]]; then
        scaffold_module "$mod" "$category" "$label"
    else
        scaffold_module "$mod"
    fi
    return $?
}

# cli_v2_completions [shell]
cli_v2_completions() {
    case "${1:-}" in
        "")      install_completions ;;
        bash|zsh|fish) UXS_COMPLETIONS_SHELL="$1" install_completions ;;
        *) error "未知 shell: $1（支持 bash / zsh / fish）"; exit "$SK_USAGE" ;;
    esac
    return $?
}

# cli_v2_cli [install|uninstall]
cli_v2_cli() {
    case "${1:-}" in
        "")        install_cli ;;
        install)   install_cli ;;
        uninstall) uninstall_cli ;;
        *) error "未知子命令: $1（支持 install / uninstall）"; exit "$SK_USAGE" ;;
    esac
    return $?
}
```

**已知取舍（明确记录，不留给实施者猜）**

- **不删 `install.sh` 文件名**：仓库入口仍是 `install.sh`，`bootstrap.sh` 与全部文档都指向它；重命名会引发远超 Stage 2 的改动面。v2 指的是**命令面**，不是文件名。全局命令 `uxs`（由 `uxs cli install` 生成）一直是这个入口的包装。
- **保留 `UNIX_SCRIPT_SHOW_ALL=1` 环境变量**：`--all` 是它的命令行形式，两者都生效；环境变量是既有逃生口，删它会让用户脚本全断。
- **模块直通保留 `uxs <mod> [subcmd...]`**：78 个模块有额外子命令（`bun mirror`/`clash start`/`sys-setup all`），新命令矩阵里没有它们的位置 ⇒ 直通是唯一不失功能的落法。已核对 78 个模块名与保留词表**无一冲突**。
- **`uxs --list-all` = `uxs list --full --all`**：全量 TSV 清单，给 AI agent 一次拿全；与 `uxs list --full`（本机适用）分工明确。
